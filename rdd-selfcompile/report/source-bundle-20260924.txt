async function handleServerMessage(msg) {
  const { requestId, action, payload } = msg;
  wsLog(`鏀跺埌鎸囦护: ${action} platform: ${payload?.platform}`);

  try {
    const platform = payload?.platform || "doubao";
    const platformConfig = PLATFORM_URLS[platform];
    if (!platformConfig) {
      sendResponse(requestId, false, null, `涓嶆敮鎸佺殑骞冲彴: ${platform}`);
      return;
    }

    let tab;

    // send_message 鏃剁敤 sessionUrl 绮剧‘瀹氫綅 tab
    if (action === "send_message" && payload?.sessionUrl) {
      const allTabs = await chrome.tabs.query({});
      tab = allTabs.find(t => t.url && t.url.startsWith(payload.sessionUrl));
      if (!tab) {
        // sessionUrl 瀵瑰簲鐨?tab 鍙兘琚叧浜嗭紝灏濊瘯閲嶆柊鎵撳紑
        wsLog(`浼氳瘽椤甸潰宸插叧闂紝閲嶆柊鎵撳紑: ${payload.sessionUrl}`);
        tab = await openChatTab(payload.sessionUrl);
      }
    } else {
      // new_session 鏃舵壘鍒版垨鎵撳紑骞冲彴椤甸潰
      tab = await findChatTab(platformConfig.pattern);
      if (!tab) {
        wsLog(`鏈壘鍒?${platform} 椤甸潰锛岃嚜鍔ㄦ墦寮€...`);
        tab = await openChatTab(platformConfig.openUrl);
      }
    }

    if (!tab) {
      sendResponse(requestId, false, null, "鏃犳硶鎵撳紑 AI 鑱婂ぉ椤甸潰");
      return;
    }

    // 纭繚 tab 澶勪簬娲昏穬鐘舵€?
    await chrome.tabs.update(tab.id, { active: true });

    // 绛夊緟 content script 灏辩华
    await waitForContentScript(tab.id);

    // 杞彂缁?content script 澶勭悊锛坆fcache 瀵艰嚧鐨勬秷鎭€氶亾澶辨晥浼氳嚜鍔ㄥ埛鏂伴噸璇曪級
    const response = await sendMessageWithBfcacheRetry(tab.id, {
      requestId,
      action,
      payload,
    });

    // 濡傛灉 content script 杩斿洖浜嗛敊璇?
    if (response && response.error) {
      sendResponse(requestId, false, null, response.error);
    } else {
      sendResponse(requestId, true, response);
    }
  } catch (e) {
    sendResponse(requestId, false, null, e.message);
  }
}

// "back/forward cache" / "message channel is closed"锛屽埛鏂伴〉闈㈤噸璇曞嵆鍙仮澶?
async function sendMessageWithBfcacheRetry(tabId, payload, retries = 2) {
  for (let i = 0; i <= retries; i++) {
    try {
      return await chrome.tabs.sendMessage(tabId, payload);
    } catch (e) {
      const msg = String((e && e.message) || e) || "";
      const bfcache = /back\/forward cache|message channel is closed|Receiving end does not exist/i.test(msg);
      if (!bfcache) throw e;
      if (i === retries) throw e;
      wsLog(`bfcache 閫氶亾澶辨晥锛屽埛鏂伴〉闈㈤噸璇?(${i + 1}/${retries})...`);
      await chrome.tabs.reload(tabId);
      await new Promise((r) => setTimeout(r, 3000));
      await waitForContentScript(tabId);
    }
  }
}

function deepMerge(base, override) {
  const out = { ...(base || {}) };
  if (!override) return out;
  for (const k of Object.keys(override)) {
    if (out[k] && typeof out[k] === "object" && typeof override[k] === "object"
        && !Array.isArray(out[k]) && !Array.isArray(override[k])) {
      out[k] = deepMerge(out[k], override[k]);
    } else {
      out[k] = override[k];
    }
  }
  return out;
}

function extractMarkdown(element) {
  let md = "";
  for (const node of element.childNodes) {
    if (node.nodeType === Node.TEXT_NODE) { md += node.textContent; continue; }
    if (node.nodeType !== Node.ELEMENT_NODE) continue;
    const tag = node.tagName.toLowerCase();
    switch (tag) {
      case "p": md += extractMarkdown(node) + "\n\n"; break;
      case "br": md += "\n"; break;
      case "strong": case "b": md += `**${extractMarkdown(node)}**`; break;
      case "em": case "i": md += `*${extractMarkdown(node)}*`; break;
      case "code":
        if (node.parentElement && node.parentElement.tagName.toLowerCase() === "pre") break;
        md += `\`${node.textContent}\``; break;
      case "pre": {
        const codeEl = node.querySelector("code");
        const codeText = codeEl ? codeEl.textContent : node.textContent;
        const langClass = codeEl?.className?.match(/language-(\w+)/);
        const lang = langClass ? langClass[1] : "";
        md += `\n\`\`\`${lang}\n${codeText}\n\`\`\`\n\n`; break;
      }
      case "ul":
        for (const li of node.querySelectorAll(":scope > li")) md += `- ${extractMarkdown(li)}\n`;
        md += "\n"; break;
      case "ol": {
        let i = 1;
        for (const li of node.querySelectorAll(":scope > li")) { md += `${i}. ${extractMarkdown(li)}\n`; i++; }
        md += "\n"; break;
      }
      case "h1": md += `# ${extractMarkdown(node)}\n\n`; break;
      case "h2": md += `## ${extractMarkdown(node)}\n\n`; break;
      case "h3": md += `### ${extractMarkdown(node)}\n\n`; break;
      case "h4": md += `#### ${extractMarkdown(node)}\n\n`; break;
      case "blockquote":
        md += extractMarkdown(node).split("\n").map(l => `> ${l}`).join("\n") + "\n\n"; break;
      case "a": md += `[${extractMarkdown(node)}](${node.href})`; break;
      case "table": {
        const rows = node.querySelectorAll("tr");
        rows.forEach((row, idx) => {
          const cells = row.querySelectorAll("th, td");
          md += "| " + Array.from(cells).map(c => c.textContent.trim()).join(" | ") + " |\n";
          if (idx === 0) md += "| " + Array.from(cells).map(() => "---").join(" | ") + " |\n";
        });
        md += "\n"; break;
      }
      default: md += extractMarkdown(node);
    }
  }
  return md;
}

async function startNewChat(config) {

function parseShortcut(str) {
  const parts = str.toLowerCase().split("+").map(s => s.trim());
  return {
    key: parts.filter(p => !["shift", "meta", "ctrl", "alt", "cmd", "command", "control"].includes(p))[0] || "",

    meta: parts.includes("meta") || parts.includes("cmd") || parts.includes("command"),
    ctrl: parts.includes("ctrl") || parts.includes("control"),
    alt: parts.includes("alt"),
  };
}

function sleep(ms) { return new Promise((resolve) => setTimeout(resolve, ms)); }

function waitForElement(selector, timeoutMs = 10000) {
  return new Promise((resolve, reject) => {
    const el = document.querySelector(selector);
    if (el) { resolve(el); return; }
    const observer = new MutationObserver(() => {
      const el = document.querySelector(selector);
      if (el) { observer.disconnect(); resolve(el); }
    });
    observer.observe(document.body, { childList: true, subtree: true });
    setTimeout(() => { observer.disconnect(); reject(new Error(`绛夊緟鍏冪礌瓒呮椂: ${selector}`)); }, timeoutMs);
  });
}

function waitForElementGone(selector, timeoutMs = 1800000) {
  return new Promise((resolve, reject) => {
    const check = () => !document.querySelector(selector);
    if (check()) { resolve(); return; }
    const observer = new MutationObserver(() => { if (check()) { observer.disconnect(); resolve(); } });
    observer.observe(document.body, { childList: true, subtree: true });
    setTimeout(() => { observer.disconnect(); if (check()) resolve(); else reject(new Error("绛夊緟鐢熸垚瀹屾瘯瓒呮椂")); }, timeoutMs);
  });
}

  function testInput(selector, method) {
    if (!selector) { log("璇峰～鍏ヨ緭鍏ユ閫夋嫨鍣?); return; }
    const target = document.querySelector(selector);
    if (!target) { log(`鎵句笉鍒板厓绱? ${selector}`); return; }

    const testText = "Hello from AI Bridge!";
    target.focus();

    switch (method) {
      case "react-setter": {
        const setter = Object.getOwnPropertyDescriptor(
          window.HTMLTextAreaElement.prototype, 'value'
        )?.set || Object.getOwnPropertyDescriptor(
          window.HTMLInputElement.prototype, 'value'
        )?.set;
        if (setter) {
          setter.call(target, testText);
          target.dispatchEvent(new Event('input', { bubbles: true }));
          target.dispatchEvent(new Event('change', { bubbles: true }));
          log(`React Setter: 宸茶緭鍏?"${testText}"`);
        } else {
          log("React Setter: 鏃犳硶鑾峰彇 value setter");
        }
        break;
      }
      case "exec-command":
        document.execCommand('insertText', false, testText);
        log(`execCommand: 宸茶緭鍏?"${testText}"`);
        break;
      case "paste": {
        const cd = new DataTransfer();
        cd.setData("text/plain", testText);
        target.dispatchEvent(new ClipboardEvent("paste", { bubbles: true, cancelable: true, clipboardData: cd }));
        log(`Paste: 宸茶緭鍏?"${testText}"`);
        break;
      }
      case "input-event":
        target.dispatchEvent(new InputEvent('beforeinput', { bubbles: true, cancelable: true, inputType: 'insertText', data: testText }));
        target.dispatchEvent(new InputEvent('input', { bubbles: true, cancelable: false, inputType: 'insertText', data: testText }));
        log(`InputEvent: 宸茶緭鍏?"${testText}"`);
        break;
    }
  }

    const socket = getSocket();
    if (socket && socket.readyState === 1) {
      resolve();
      return;
    }
    log("[ws-send] 鏈繛鎺ワ紝绛夊緟...");
    const startTime = Date.now();
    const check = setInterval(() => {
      const s = getSocket();
      if (s && s.readyState === 1) {
        clearInterval(check);
        log("[ws-send] 宸茶繛鎺?);
        resolve();
      } else if (Date.now() - startTime > timeoutMs) {
        clearInterval(check);
        reject(new Error("绛夊緟鎻掍欢杩炴帴瓒呮椂锛?0绉掞級銆傝妫€鏌ワ細1) Chrome 鎻掍欢宸插畨瑁?2) 鎻掍欢璁剧疆涓凡寮€鍚繛鎺?3) Token 姝ｇ‘"));
      }
    }, 1000);
  });
}

async function sendToExtension(action, payload, timeoutMs = 1800000) {
  await waitForConnection();

  return new Promise((resolve, reject) => {
    const requestId = Math.random().toString(36).substring(2, 10);
    log(`[ws-send] action=${action} requestId=${requestId} platform=${payload?.platform} timeout=${timeoutMs/1000}s`);

    const timeout = setTimeout(() => {
      pendingRequests.delete(requestId);
      log(`[ws-send] 瓒呮椂: requestId=${requestId} action=${action}`);
      reject(new Error(`璇锋眰瓒呮椂锛?{timeoutMs / 1000}绉掞級`));
    }, timeoutMs);

    pendingRequests.set(requestId, { resolve, reject, timeout });

    getSocket().send(JSON.stringify({
      requestId,
      action,
      payload,
    }));
    log(`[ws-send] 宸插彂閫侊紝褰撳墠 pending 鏁? ${pendingRequests.size}`);
  });
}

function buildSections() {

    const body = fs.readFileSync(p, "utf8");
    sections.push({ rel, text: `===== FILE: ${rel} (${body.length} chars) =====\n${body}\n===== END ${rel} =====\n` });
  }
  return sections;
}

function packChunks(sections) {
  const chunks = [];
  let cur = "", n = 0;
  for (const s of sections) {
    if (cur && n + s.text.length > CHUNK_TARGET) { chunks.push(cur); cur = ""; n = 0; }
    cur += s.text; n += s.text.length;
  }
  if (cur.trim()) chunks.push(cur);
  return chunks;
}

function Get-OpenCodeWindowHandle {
  Add-Type @"

using System.Text;
using System.Runtime.InteropServices;
public class OcWin {
  [DllImport("user32.dll")] public static extern bool EnumWindows(EnumProc cb, IntPtr l);
  public delegate bool EnumProc(IntPtr h, IntPtr l);
  [DllImport("user32.dll")] public static extern int GetWindowText(IntPtr h, StringBuilder s, int m);
  [DllImport("user32.dll")] public static extern bool IsWindowVisible(IntPtr h);
  [DllImport("user32.dll")] public static extern uint GetWindowThreadProcessId(IntPtr h, out uint p);

  [DllImport("user32.dll")] public static extern bool SetForegroundWindow(IntPtr h);
  public static IntPtr Find() {
    IntPtr found = IntPtr.Zero;
    EnumWindows((h, l) => {
      if (!IsWindowVisible(h)) return true;
      StringBuilder sb = new StringBuilder(512);

      if (sb.ToString().ToLowerInvariant().Contains("opencode")) { found = h; return false; }

function Send-Nudge {
  param(
    [string]$Message = "continue working on current task, do not stop; if you hit an error log it and skip it.",
    [switch]$DryRun
  )
  Add-Type -AssemblyName System.Windows.Forms
  $h = Get-OpenCodeWindowHandle
  if ($h -eq [IntPtr]::Zero) {
    Write-Log "NUDGE fail: opencode window not found"
    return $false
  }
  $sh = New-Object -ComObject WScript.Shell
  $tryForeground = [OcWin]::Fg()
  if ($tryForeground -ne $h) {
    [OcWin]::Activate($h) | Out-Null
    Start-Sleep -Milliseconds 400
    $tryForeground = [OcWin]::Fg()
  }
  if ($tryForeground -ne $h) {
    Write-Log "NUDGE skip: foreground=$tryForeground != target $h (another window owns focus, e.g. MC). Try again next round."
    return $false
  }
  Start-Sleep -Milliseconds 300
  if ($DryRun) {
    Write-Log "NUDGE dryrun target=$h fg=$tryForeground (would send: $Message)"
    return $true
  }
  [System.Windows.Forms.Clipboard]::SetText($Message)
  Start-Sleep -Milliseconds 200
  [System.Windows.Forms.SendKeys]::SendWait("^v")
  Start-Sleep -Milliseconds 150
  [System.Windows.Forms.SendKeys]::SendWait("{ENTER}")
  Write-Log ("NUDGE sent to window " + $h)
  return $true
}

function Read-Queue {

function Do-One-Round {
  Write-Log "ROUND begin"
  $q = Read-Queue
  if ($null -eq $q) {
    Write-Log "ROUND queue empty(no file) - heartbeat only"
    return
  }
  $task = $null
  try {
    if ($q.tasks -and $q.tasks.Count -gt 0) { $task = $q.tasks[0].text }
  } catch { $task = $null }
  if ($null -ne $task -and $task.Length -gt 0) {
    $reqId = "req-{0:yyyyMMddHHmmss}" -f (Get-Date)
    $body = @{
      ts    = (Get-Date).ToUniversalTime().ToString("o")
      reqId = $reqId
      task  = $task
      source = "every-10min-drive"
    } | ConvertTo-Json -Depth 4
    $reqFile = Join-Path $global:BP_DropDir ($reqId + ".json")
    Set-Content -Path $reqFile -Value $body -Encoding UTF8
    Write-Log ("REQUEST dropped " + $reqFile)
  } else {
    Write-Log "ROUND queue no task - heartbeat only"
  }
}
